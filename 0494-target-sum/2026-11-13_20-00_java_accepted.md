class Solution {
    public int findTargetSumWays(int[] nums, int target) {
        int n=nums.length;
        int sum=0;
        for(int i=0;i<n;i++){

            sum=sum+nums[i];
        }
        if(Math.abs(target)>sum){
            return 0;
        }
        int requiredtarget= (sum+target)/2;
        if((sum + target)  % 2!=0){
            return 0;
        }
        int dp[][]=new int[n+1][requiredtarget+1];
        for(int i=0;i<=n;i++){
            dp[i][0]=1;
        }
        // Base case
        dp[0][0] = 1;
        for(int i=1;i<=n;i++){
            for(int j=0;j<=requiredtarget;j++){
                dp[i][j]=dp[i-1][j];
                if(nums[i-1]<=j){
                    dp[i][j]+= dp[i-1][j-nums[i-1]];
                }
                
            }
        }
        return dp[n][requiredtarget];
        
        
    }
}

// Logic:

// Convert the problem into a Count Subset Sum (0/1 Knapsack) problem.

// Let:
// - P = Sum of positive elements.
// - N = Sum of negative elements (absolute value).
// - S = Total sum of the array.
// We know:

// P - N = target

// P + N = S

// Adding both equations:

// 2P = S + target

// Therefore:

// P = (S + target) / 2

// Now, the problem becomes finding the number of subsets whose sum equals (S + target) / 2.

// Approach:
// 1. Calculate the total sum of all elements in the array.
// 2. If Math.abs(target) > sum, return 0 because the target is impossible.
// 3. If (sum + target) % 2 != 0, return 0 because the required subset sum is not an integer.
// 4. Calculate requiredSum = (sum + target) / 2.
// 5. Create a 2D DP array: dp[n+1][requiredSum+1].
// 6. Define dp[i][j] as the number of ways to make sum j using the first i elements.
// 7. Initialize base cases:
//    - dp[0][0] = 1 (one way to make sum 0: select nothing).
//    - dp[0][j] = 0 for j > 0.
// 8. Iterate through each element and each possible sum.
// 9. Apply the DP transition:
//    - Exclude current element: dp[i-1][j].
