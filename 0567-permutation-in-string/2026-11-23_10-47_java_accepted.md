class Solution {
    public boolean checkInclusion(String s1, String s2) {
        int n1=s1.length(); //Sliding window length should be length of s1;
        int n2=s2.length();
        int i=0;
        int k=n1;
        int j=0;
        HashMap<Character, Integer> map= new HashMap<>();
        for(int l=0;l<n1;l++){
            char ch=s1.charAt(l);
            if(map.containsKey(ch)){
                map.put(ch, map.get(ch)+1);
            }else{
                map.put(ch,1);
            }
        }
        while(j<n2){
            char ch=s2.charAt(j);
            map.put(ch, map.getOrDefault(ch,0)-1);
            if(j-i+1>k){
                char ch2=s2.charAt(i);
                map.put(ch2, map.getOrDefault(ch2,0)+1);
                i++;
            }
            if(j-i+1==k){
                boolean isValid=true;
                for(int value: map.values()){
                    if(value!=0){
                        isValid=false;
                        break;

                    }
                }
            if(isValid){
                return true;
            }
            }

            j++;
            

        }
        return false;

        
    }
}
