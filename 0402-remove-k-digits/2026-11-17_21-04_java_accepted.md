## Solution

```java
class Solution {
    public String removeKdigits(String num, int k) {
        Stack<Character> st=new Stack<>();
        int  n=num.length();

        if(n==k){
            return "0";
        }

        for(int i=0;i<n;i++){
            while(k>0 && !st.empty() && num.charAt(i)-'0'< st.peek()-'0'){
                st.pop();
                k=k-1;
            }
            st.push(num.charAt(i));
       
        }

        while (k > 0) {
            st.pop();
            k--;
        }

        StringBuilder sb= new StringBuilder();
        while(!st.empty()){
            sb.append(st.pop());
        }
        sb.reverse();

        //edge case trailing zeroes
   
        while(sb.length() > 0 && sb.charAt(0)=='0'){
            sb.deleteCharAt(0);
        }

        if (sb.length() == 0) {
            return "0";
        }

        return sb.toString();
          
    }
}
```
