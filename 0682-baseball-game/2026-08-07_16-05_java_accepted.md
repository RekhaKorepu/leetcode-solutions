## Solution

```java
class Solution {
    public int calPoints(String[] operations) {
        Stack<Integer> st = new Stack<>();
        for(int i=0; i<operations.length; i++){
            String op = operations[i];
            char c = op.charAt(0);
            if(Character.isDigit(c) || c == '-' || c == '+' && op.length() > 1){
                st.push(Integer.parseInt(operations[i]));
            }
            else if(op.equals("C")){
                st.pop();
            }
            else if(op.equals("D")){
                st.push(st.peek()*2);
            }
            else if(op.equals("+")){
                int top= st.peek();
                int second= st.get(st.size()-2);
                st.push(top+second);
            }      
        }
        int res=0;
        while(!st.empty()){
            res=res+st.pop();
        }

        return res;

        
    }
}
```
