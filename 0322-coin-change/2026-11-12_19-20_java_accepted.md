## Solution

```java
class Solution {
    public int coinChange(int[] coins, int amount) {
       int n=coins.length;
       int dp[]=new int[amount+1];
 
       Arrays.fill(dp, Integer.MAX_VALUE);
       dp[0]=0;
       for(int i=1;i<=amount;i++){
         for(int coin: coins){
            if(coin<=i && dp[i-coin]!=Integer.MAX_VALUE){     
                dp[i]= Math.min(dp[i], 1+dp[i-coin]); //dp[i]==> exclude and 
                // 1+dp[i-coin] means include (1 is for counting that coin and min coins required for making remaining amount)
            }

          }
       }
       return dp[amount]!=Integer.MAX_VALUE ? dp[amount]: -1;
        
    }
}
```
