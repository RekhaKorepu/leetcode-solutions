 *         this.right = right;
 *     }

class Solution {
    public void flatten(TreeNode root) {
        Stack<TreeNode> st= new Stack<>();
        //we have to push in order root->right and left
        if(root==null){
            return;
        }
        st.push(root);
        while(!st.empty()){
        TreeNode curr= st.pop();
        if(curr.right!=null){
            st.push(curr.right);
        }
        if(curr.left!=null){
            st.push(curr.left);
        }
        if(!st.empty()){
          curr.right=st.peek();
        }

        curr.left= null;
    }
        
    }
}
