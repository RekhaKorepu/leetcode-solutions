## Solution

```java
/**
 * Definition for singly-linked list.
 * class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode(int x) {
 *         val = x;
 *         next = null;
 *     }
 * }
 */

//  Remember this interview pattern
// - Cycle detection: Slow moves 1 step, fast moves 2 steps.
// - Cycle starting point: Reset one pointer to head.
// - Move both pointers: One step at a time.
// - Return: The node where both pointers meet.
// Complexity:
// - Time: O(n)
// - Space: O(1)
public class Solution {
    public ListNode detectCycle(ListNode head) {
        
        ListNode slow = head;
        ListNode fast=head;

        if(head==null){
            return head;
        }

        while(fast!=null && fast.next!=null){
            slow=slow.next;
            fast=fast.next.next;
            if(slow==fast){
                slow =head;
                while(slow!=fast){
                    slow=slow.next;
                    fast=fast.next;
                }
                return slow;

            }

        }

        return null;

        
    }
    
}
```
